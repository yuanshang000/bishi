### 3.1 SQLAlchemy ORM模型

#### 3.1.1 User模型 (models/user.py)

```python
from sqlalchemy import Column, Integer, String, Enum, DateTime, Boolean
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from app.core.database import Base
import enum

class UserRole(str, enum.Enum):
    """用户角色枚举"""
    STUDENT = "student"
    TEACHER = "teacher"

class User(Base):
    """用户模型"""
    __tablename__ = "users"

    # 主键
    id = Column(Integer, primary_key=True, index=True, autoincrement=True)

    # 基本信息
    username = Column(String(50), unique=True, nullable=False, index=True)
    password_hash = Column(String(255), nullable=False)
    role = Column(Enum(UserRole), nullable=False, default=UserRole.STUDENT, index=True)

    # 可选信息
    email = Column(String(100), nullable=True)
    real_name = Column(String(50), nullable=True)

    # 时间戳
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())
    last_login = Column(DateTime(timezone=True), nullable=True)

    # 状态标志
    is_active = Column(Boolean, default=True, nullable=False)

    # 关系
    quiz_records = relationship("QuizRecord", back_populates="user", cascade="all, delete-orphan")
    chat_histories = relationship("ChatHistory", back_populates="user", cascade="all, delete-orphan")
    created_chapters = relationship("Chapter", foreign_keys="Chapter.created_by", back_populates="creator")
    uploaded_documents = relationship("CourseDocument", foreign_keys="CourseDocument.uploaded_by", back_populates="uploader")

    def __repr__(self):
        return f"<User(id={self.id}, username={self.username}, role={self.role})>"

    @property
    def is_teacher(self) -> bool:
        """判断是否为教师"""
        return self.role == UserRole.TEACHER

    @property
    def is_student(self) -> bool:
        """判断是否为学生"""
        return self.role == UserRole.STUDENT
```

#### 3.1.2 Chapter模型 (models/chapter.py)

```python
from sqlalchemy import Column, Integer, String, Text, DateTime, ForeignKey
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from app.core.database import Base

class Chapter(Base):
    """章节模型"""
    __tablename__ = "chapters"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String(200), nullable=False)
    sequence_order = Column(Integer, nullable=False, index=True)
    parent_id = Column(Integer, ForeignKey("chapters.id"), nullable=True, index=True)
    description = Column(Text, nullable=True)

    # 时间戳
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    # 创建人
    created_by = Column(Integer, ForeignKey("users.id"), nullable=True)

    # 关系
    creator = relationship("User", foreign_keys=[created_by], back_populates="created_chapters")
    documents = relationship("CourseDocument", back_populates="chapter", cascade="all, delete-orphan")
    quiz_records = relationship("QuizRecord", back_populates="chapter")
    parent = relationship("Chapter", remote_side=[id], backref="children")

    def __repr__(self):
        return f"<Chapter(id={self.id}, title={self.title}, sequence={self.sequence_order})>"

    @property
    def document_count(self) -> int:
        """获取教案数量"""
        return len(self.documents)
```

#### 3.1.4 QuizRecord模型 (models/quiz_record.py)

```python
from sqlalchemy import Column, Integer, String, Text, Boolean, DateTime, Enum, ForeignKey, JSON
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from app.core.database import Base
import enum

class QuizType(str, enum.Enum):
    """练习类型枚举"""
    CHAPTER_QUIZ = "chapter_quiz"
    FOLLOWUP_QUIZ = "followup_quiz"
    ADAPTIVE_QUIZ = "adaptive_quiz"

class QuizRecord(Base):
    """练习记录模型"""
    __tablename__ = "quiz_records"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"), nullable=False, index=True)
    chapter_id = Column(Integer, ForeignKey("chapters.id"), nullable=True, index=True)
    quiz_type = Column(Enum(QuizType), nullable=False, index=True)

    # 题目内容
    question_text = Column(Text, nullable=False)
    options = Column(JSON, nullable=False)  # 存储为JSON: {"A": "选项A", "B": "选项B", ...}
    correct_answer = Column(String(10), nullable=False)

    # 答题结果
    user_answer = Column(String(10), nullable=True)
    is_correct = Column(Boolean, nullable=True)
    score = Column(Integer, default=0)

    # AI生成的解析
    explanation = Column(Text, nullable=True)

    # 知识点标签
    knowledge_tags = Column(JSON, nullable=True)  # ["tag1", "tag2"]

    # 时间
    created_at = Column(DateTime(timezone=True), server_default=func.now(), index=True)
    time_spent = Column(Integer, nullable=True)  # 秒

    # 关系
    user = relationship("User", back_populates="quiz_records")
    chapter = relationship("Chapter", back_populates="quiz_records")

    def __repr__(self):
        return f"<QuizRecord(id={self.id}, user_id={self.user_id}, correct={self.is_correct})>"

    @property
    def is_answered(self) -> bool:
        """判断是否已答题"""
        return self.user_answer is not None
```

### 3.2 Pydantic验证模型

#### 3.2.1 User Schema (schemas/user.py)

```python
from pydantic import BaseModel, EmailStr, Field, validator
from typing import Optional
from datetime import datetime
from app.models.user import UserRole

# 注册请求
class UserRegisterRequest(BaseModel):
    username: str = Field(..., min_length=3, max_length=50, description="用户名")
    password: str = Field(..., min_length=6, max_length=100, description="密码")
    role: UserRole = Field(default=UserRole.STUDENT, description="角色")
    email: Optional[EmailStr] = Field(None, description="邮箱")
    real_name: Optional[str] = Field(None, max_length=50, description="真实姓名")

    @validator("username")
    def validate_username(cls, v):
        if not v.isalnum() and "_" not in v:
            raise ValueError("用户名只能包含字母、数字和下划线")
        return v

# 登录请求
class UserLoginRequest(BaseModel):
    username: str = Field(..., description="用户名")
    password: str = Field(..., description="密码")

# Token响应
class TokenResponse(BaseModel):
    token: str
    token_type: str = "Bearer"
    expires_in: int = 86400  # 24小时
    user: "UserResponse"

# 用户信息响应
class UserResponse(BaseModel):
    user_id: int
    username: str
    role: UserRole
    email: Optional[str]
    real_name: Optional[str]
    created_at: datetime
    last_login: Optional[datetime]

    class Config:
        orm_mode = True
```

#### 3.2.2 Quiz Schema (schemas/quiz.py)

```python
from pydantic import BaseModel, Field
from typing import List, Dict, Optional
from datetime import datetime
from app.models.quiz_record import QuizType

# 生成题目请求
class GenerateQuizRequest(BaseModel):
    chapter_id: int = Field(..., description="章节ID")
    count: int = Field(default=5, ge=1, le=20, description="题目数量")
    difficulty: str = Field(default="medium", regex="^(easy|medium|hard)$", description="难度")

# 单个题目
class Question(BaseModel):
    question_id: Optional[int] = None
    question_text: str
    options: Dict[str, str]  # {"A": "选项A", "B": "选项B", ...}
    correct_answer: Optional[str] = None  # 生成时不返回
    knowledge_tags: List[str]

# 生成题目响应
class GenerateQuizResponse(BaseModel):
    quiz_id: str
    chapter_id: int
    questions: List[Question]
    total_questions: int

# 提交答案请求
class SubmitAnswerRequest(BaseModel):
    chapter_id: int
    quiz_type: QuizType
    answers: List["AnswerItem"]
```

### 5.1 文本分块算法 (utils/text_splitter.py)

```python
from typing import List
import re

class TextSplitter:
    """文本分块工具"""

    def __init__(
        self,
        chunk_size: int = 800,
        chunk_overlap: int = 100,
        separator: str = "\n\n"
    ):
        """
        初始化分块器

        Args:
            chunk_size: 每个分块的目标字符数
            chunk_overlap: 分块间重叠的字符数
            separator: 优先分割符(段落分隔符)
        """
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.separator = separator

    def split_text(self, text: str) -> List[str]:
        """
        将长文本分割成多个块

        算法步骤:
        1. 按段落分隔符分割
        2. 合并小段落直到达到目标大小
        3. 添加重叠区域

        Args:
            text: 输入文本

        Returns:
            分块后的文本列表
        """
        # 清理文本
        text = self._clean_text(text)

        # 按段落分割
        paragraphs = text.split(self.separator)
        paragraphs = [p.strip() for p in paragraphs if p.strip()]

        chunks = []
        current_chunk = ""

        for paragraph in paragraphs:
            # 如果段落本身超过chunk_size,需要进一步分割
            if len(paragraph) > self.chunk_size:
                # 先添加当前累积的chunk
                if current_chunk:
                    chunks.append(current_chunk.strip())
                    current_chunk = ""

                # 分割超长段落
                sub_chunks = self._split_long_paragraph(paragraph)
                chunks.extend(sub_chunks)
            else:
                # 尝试添加到当前chunk
                if len(current_chunk) + len(paragraph) + len(self.separator) <= self.chunk_size:
                    current_chunk += paragraph + self.separator
                else:
                    # 当前chunk已满,保存并开始新chunk
                    if current_chunk:
                        chunks.append(current_chunk.strip())

                    # 添加重叠内容
                    overlap_text = self._get_overlap_text(current_chunk)
                    current_chunk = overlap_text + paragraph + self.separator

        # 添加最后一个chunk
        if current_chunk:
            chunks.append(current_chunk.strip())

        return chunks

    def _clean_text(self, text: str) -> str:
        """清理文本"""
        # 移除多余空白
        text = re.sub(r'\s+', ' ', text)
        # 恢复段落分隔
        text = text.replace(' \n ', '\n\n')
        return text.strip()

    def _split_long_paragraph(self, paragraph: str) -> List[str]:
        """分割超长段落"""
        chunks = []
        sentences = re.split(r'([。!?])', paragraph)

        current = ""
        for i in range(0, len(sentences), 2):
            sentence = sentences[i] + (sentences[i+1] if i+1 < len(sentences) else "")

            if len(current) + len(sentence) <= self.chunk_size:
                current += sentence
            else:
                if current:
                    chunks.append(current.strip())
                current = sentence

        if current:
            chunks.append(current.strip())

        return chunks

    def _get_overlap_text(self, text: str) -> str:
        """获取重叠文本"""
        if len(text) <= self.chunk_overlap:
            return text
        return text[-self.chunk_overlap:] + " "
```

**算法复杂度:**
- 时间复杂度: O(n),其中n为文本长度
- 空间复杂度: O(n)

### 5.2 向量化批处理算法 (services/document_service.py)

```python
async def batch_embed_and_store(
    self,
    document_id: int,
    chunks: List[str],
    chapter_id: int,
    file_name: str
) -> int:
    """
    批量向量化并存储

    算法设计:
    1. 将chunks分批(每批10个)
    2. 并发调用Embedding API
    3. 批量写入ChromaDB
    4. 更新数据库索引状态

    Args:
        document_id: 文档ID
        chunks: 文本分块列表
        chapter_id: 章节ID
        file_name: 文件名

    Returns:
        成功向量化的数量
    """
    BATCH_SIZE = 10
    total_chunks = len(chunks)
    success_count = 0

    for i in range(0, total_chunks, BATCH_SIZE):
        batch = chunks[i:i + BATCH_SIZE]

        try:
            # 并发生成向量
            embeddings = await self._batch_generate_embeddings(batch)

            # 构造metadata
            metadatas = [
                {
                    "chapter_id": chapter_id,
                    "document_id": document_id,
                    "chunk_index": i + idx,
                    "source_file": file_name
                }
                for idx in range(len(batch))
            ]

            # 生成唯一ID
            ids = [f"doc_{document_id}_chunk_{i + idx}" for idx in range(len(batch))]

            # 存储到ChromaDB
            self.rag_service.add_documents(
                documents=batch,
                embeddings=embeddings,
                metadatas=metadatas,
                ids=ids
            )

            success_count += len(batch)
            logger.info(f"批次 {i // BATCH_SIZE + 1} 向量化成功: {len(batch)} 个片段")

        except Exception as e:
            logger.error(f"批次 {i // BATCH_SIZE + 1} 向量化失败: {str(e)}")
            # 继续处理下一批次

    return success_count

async def _batch_generate_embeddings(self, texts: List[str]) -> List[List[float]]:
    """并发生成向量"""
    import asyncio
    tasks = [self.llm_service.generate_embedding(text) for text in texts]
    embeddings = await asyncio.gather(*tasks)
    return embeddings
```

**算法优势:**
- 批量处理减少API调用次数
- 并发执行提高处理速度
- 失败隔离,部分失败不影响整体

### 5.3 章节溯源算法 (services/chat_service.py)

```python
def extract_related_chapter(self, retrieval_results: Dict) -> Optional[int]:
    """
    从检索结果中提取最相关的章节

    算法:
    1. 提取所有chunk的chapter_id
    2. 计算每个chapter_id的出现次数
    3. 返回出现次数最多的chapter_id

    Args:
        retrieval_results: ChromaDB检索结果

    Returns:
        最相关的章节ID
    """
    from collections import Counter

    metadatas = retrieval_results.get('metadatas', [[]])[0]

    if not metadatas:
        return None

    chapter_ids = [meta.get('chapter_id') for meta in metadatas if 'chapter_id' in meta]

    if not chapter_ids:
        return None

    # 统计出现频率
    counter = Counter(chapter_ids)
    most_common_chapter_id, count = counter.most_common(1)[0]

    logger.info(f"章节溯源: chapter_id={most_common_chapter_id}, 出现次数={count}/{len(chapter_ids)}")

    return most_common_chapter_id
```

#### 6.4.1 MCP Server主程序 (mcp_server/server.py)

```python
"""
MCP Server - 数据工具提供者

设计理念：
- 单一职责：仅负责数据获取，不做业务逻辑
- 无状态：每次调用独立
- 透明性：原样返回数据库数据
"""

import asyncio
import json
from typing import Dict, Any, List
from mcp.server import MCPServer
from mcp.types import Tool, TextContent
from mcp_server.config import MCPConfig
from mcp_server.tools.fetch_learning_data import FetchLearningDataTool
import logging

logger = logging.getLogger(__name__)

class CollegeEnglishMCPServer:
    """大学英语智慧助手MCP服务器"""

    def __init__(self):
        self.server = MCPServer(name="college-english-assistant")
        self.config = MCPConfig()
        self.tools: Dict[str, Any] = {}

        # 注册工具
        self._register_tools()

        # 设置handlers
        self._setup_handlers()

    def _register_tools(self):
        """注册MCP工具"""
        # 注册fetch_learning_data工具
        fetch_tool = FetchLearningDataTool(self.config.get_db_connection())
        self.tools["fetch_learning_data"] = fetch_tool

        logger.info(f"已注册 {len(self.tools)} 个MCP工具")

    def _setup_handlers(self):
        """设置MCP handlers"""

        @self.server.list_tools()
        async def list_tools() -> List[Tool]:
            """列出所有可用工具"""
            return [
                Tool(
                    name="fetch_learning_data",
                    description="获取学生的练习记录数据，包括章节、分数、知识点标签、答题时间等原始信息",
                    inputSchema={
                        "type": "object",
                        "properties": {
                            "user_id": {
                                "type": "integer",
                                "description": "学生ID"
                            },
                            "time_range": {
                                "type": "integer",
                                "description": "时间范围（天数），可选，默认返回全部",
                                "default": None
                            }
                        },
                        "required": ["user_id"]
                    }
                )
            ]

        @self.server.call_tool()
        async def call_tool(name: str, arguments: Dict[str, Any]) -> List[TextContent]:
            """调用工具"""
            if name not in self.tools:
                raise ValueError(f"未知工具: {name}")

            logger.info(f"调用MCP工具: {name}, 参数: {arguments}")

            try:
                # 调用工具
                result = await self.tools[name].execute(arguments)

                # 返回结果
                return [TextContent(
                    type="text",
                    text=json.dumps(result, ensure_ascii=False, indent=2)
                )]
            except Exception as e:
                logger.error(f"工具调用失败: {str(e)}")
                return [TextContent(
                    type="text",
                    text=json.dumps({
                        "error": str(e),
                        "success": False
                    }, ensure_ascii=False)
                )]

    async def run(self, transport: str = "stdio"):
        """运行MCP服务器"""
        logger.info(f"启动MCP服务器，传输方式: {transport}")

        if transport == "stdio":
            from mcp.server.stdio import stdio_server
            async with stdio_server() as streams:
                await self.server.run(
                    streams[0],
                    streams[1],
                    self.server.create_initialization_options()
                )
        else:
            raise ValueError(f"不支持的传输方式: {transport}")

def main():
    """主函数"""
    logging.basicConfig(
        level=logging.INFO,
        format='[%(asctime)s] [%(levelname)s] [%(name)s] %(message)s'
    )

    server = CollegeEnglishMCPServer()
    asyncio.run(server.run())

if __name__ == "__main__":
    main()
```

#### 6.4.3 Fetch Learning Data工具 (mcp_server/tools/fetch_learning_data.py)

```python
from typing import Dict, Any, List
from datetime import datetime, timedelta
from mcp_server.tools.base_tool import BaseTool
import logging

logger = logging.getLogger(__name__)

class FetchLearningDataTool(BaseTool):
    """获取学习数据工具"""

    async def execute(self, arguments: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行数据获取

        Args:
            arguments: {"user_id": int, "time_range": int (optional)}

        Returns:
            {"success": bool, "records": List[Dict]}
        """
        # 验证参数
        if not self.validate_arguments(arguments):
            return {
                "success": False,
                "error": "参数验证失败",
                "records": []
            }

        user_id = arguments["user_id"]
        time_range = arguments.get("time_range")

        try:
            # 构建SQL查询（原样返回数据，不做聚合）
            query = """
                SELECT
                    qr.id,
                    qr.chapter_id,
                    c.title as chapter_title,
                    qr.quiz_type,
                    qr.question_text,
                    qr.options,
                    qr.correct_answer,
                    qr.user_answer,
                    qr.is_correct,
                    qr.score,
                    qr.knowledge_tags,
                    qr.created_at,
                    qr.time_spent
                FROM quiz_records qr
                LEFT JOIN chapters c ON qr.chapter_id = c.id
                WHERE qr.user_id = %s
            """

            params = [user_id]

            # 添加时间范围过滤
            if time_range:
                cutoff_date = datetime.now() - timedelta(days=time_range)
                query += " AND qr.created_at >= %s"
                params.append(cutoff_date)

            query += " ORDER BY qr.created_at DESC"

            # 执行查询
            cursor = self.db.cursor(dictionary=True)
            cursor.execute(query, params)
            records = cursor.fetchall()
            cursor.close()

            # 转换日期为字符串
            for record in records:
                if record.get('created_at'):
                    record['created_at'] = record['created_at'].isoformat()

            logger.info(f"获取学习数据成功: user_id={user_id}, 记录数={len(records)}")

            return {
                "success": True,
                "user_id": user_id,
                "record_count": len(records),
                "records": records
            }

        except Exception as e:
            logger.error(f"获取学习数据失败: {str(e)}")
            return {
                "success": False,
                "error": str(e),
                "records": []
            }

    def validate_arguments(self, arguments: Dict[str, Any]) -> bool:
        """验证参数"""
        if "user_id" not in arguments:
            return False

        if not isinstance(arguments["user_id"], int):
            return False

        if "time_range" in arguments and arguments["time_range"] is not None:
            if not isinstance(arguments["time_range"], int) or arguments["time_range"] < 1:
                return False

        return True
```

### 7.6 Pinia Store设计 (stores/user.ts)

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UserInfo } from '@/types/user'

export const useUserStore = defineStore('user', () => {
  // State
  const token = ref<string>(localStorage.getItem('token') || '')
  const userInfo = ref<UserInfo | null>(null)

  // Getters
  const isLoggedIn = computed(() => !!token.value)
  const isTeacher = computed(() => userInfo.value?.role === 'teacher')
  const isStudent = computed(() => userInfo.value?.role === 'student')

  // Actions
  function setToken(newToken: string) {
    token.value = newToken
    localStorage.setItem('token', newToken)
  }

  function setUser(user: UserInfo) {
    userInfo.value = user
  }

  function logout() {
    token.value = ''
    userInfo.value = null
    localStorage.removeItem('token')
  }

  function getAuthHeader() {
    return {
      Authorization: `Bearer ${token.value}`
    }
  }

  return {
    token,
    userInfo,
    isLoggedIn,
    isTeacher,
    isStudent,
    setToken,
    setUser,
    logout,
    getAuthHeader
  }
})
```

### 9.4 Nginx配置

```nginx
server {
    listen 80;
    server_name your-domain.com;

    # 前端静态文件
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
        index index.html;
    }

    # API代理
    location /api/ {
        proxy_pass http://backend:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 超时设置
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # 文件上传大小限制
    client_max_body_size 50M;

    # Gzip压缩
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml;
    gzip_min_length 1000;
}
```

### 10.5 开发规范

#### 10.5.1 Python代码规范

- 遵循PEP 8规范
- 使用类型注解
- 编写Docstring（Google风格）
- 单元测试覆盖率 > 80%
- 使用black格式化代码
- 使用pylint检查代码质量

#### 10.5.2 TypeScript代码规范

- 遵循ESLint规则
- 使用严格模式
- 组件Props必须定义类型
- 避免使用any类型
- 使用Prettier格式化代码

#### 10.5.3 Git提交规范

```
<type>(<scope>): <subject>

<body>

<footer>
```

类型：
- feat: 新功能
- fix: 修复bug
- docs: 文档修改
- style: 代码格式调整
- refactor: 重构
- test: 测试相关
- chore: 构建/配置修改

示例：
```
feat(quiz): 添加章节练习功能

- 实现题目生成API
- 完成答题判分逻辑
- 添加练习历史查询

Closes #123
```
